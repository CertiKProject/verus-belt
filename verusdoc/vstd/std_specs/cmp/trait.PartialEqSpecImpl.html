open spec fn eq_spec(&self, other: &Option<T>) -> bool

{
    match (self, other) {
        (None, None) => true,
        (Some(x), Some(y)) => x.eq_spec(y),
        _ => false,
    }
}

open spec fn eq_spec(&self, other: &Vec<U, A2>) -> bool

{
    &&& self.len() == other.len()
    &&& forall |i: int| 0 <= i < self.len() ==> self[i].eq_spec(&other[i])

}
